% ===================== 4. The objective =====================
\begin{frame}{The objective: one composite, five design qualities}
  \[
    \text{score}(\theta) \;=\; w_F\,\overline{F} \;+\; w_{F_{\max}}\,F_{\max}
    \;+\; w_R\,\tfrac{|\overline{R}-R^*|}{R^*} \;+\; w_D\,\overline{D}
    \;+\; w_T\,\tfrac{|\overline{T}-T^*|}{T^*}
  \]
  \vspace{0.2em}
  \begin{center}
  \small
  \begin{tabular}{lll}
    \toprule
    Term & Measures & Goal \\
    \midrule
    $\overline{F}$ mean fairness     & win-rate spread between players (avg.\ over matchups) & minimise \\
    $F_{\max}$ worst-pair fairness   & the \emph{worst} matchup's spread (robustness)        & minimise \\
    $\overline{R}$ length            & mean rounds per game                                  & target $60$ \\
    $\overline{D}$ draw rate         & fraction of games with no winner                      & minimise \\
    $\overline{T}$ transfer          & inter-player money moved per player-turn              & target $50$ \\
    \bottomrule
  \end{tabular}
  \end{center}
\end{frame}

% ===================== 8. Game-design insights: solvency =====================
\begin{frame}{Game-design insights}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.55\textwidth}
      \begin{itemize}\footnotesize\itemsep=0.4em
        \item \textbf{Expansion $\rightarrow$ solvency: losing condition flips.}
          \begin{itemize}\scriptsize\itemsep=0.2em
            \item Default (long games, low rents): reward owning more.
            \item GA winner ($\sim$$60$ rounds, $\sim$$1.3\times$ rent, $\sim$$1.6\times$ cost): reward staying solvent.
            \item One bad landing can drain a \$$300$ reserve.
            \item AggressiveBuilder vs.\ CashHoarder: $0.80\!\to\!0.45$ (WR).
          \end{itemize}
      \end{itemize}
    \end{column}

% ===================== 9. Game-design insights: colour groups =====================
\begin{frame}{Game-design insights}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.55\textwidth}
      \begin{itemize}\footnotesize\itemsep=0.4em
        \item \textbf{Smaller colour groups make all-in builds work.}
          \begin{itemize}\scriptsize\itemsep=0.2em
            \item Default: cheap 2-cell groups (Brown) can't win; need an expensive triplet.
            \item GA winner: $1$--$2$ cell groups $+$ $1.6\times$ cost / $1.3\times$ rent $\rightarrow$ max houses $+$ hotel pays off.
          \end{itemize}
      \end{itemize}

% ===================== 10. Game-design insights: blocking =====================
\begin{frame}{Game-design insights}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.55\textwidth}
      \begin{itemize}\footnotesize\itemsep=0.4em
        \item \textbf{Blocking opponents stops working.}
          \begin{itemize}\scriptsize\itemsep=0.2em
            \item Default: buy one cell of opponent's set to deny completion.
            \item GA winner: $1$--$2$ cell sets complete on first landing.
            \item Higher costs ($\sim$$1.6\times$) prevent hoarding properties before crossing GO.
            \item Something rent tuning alone could not do.
          \end{itemize}
      \end{itemize}

% ===================== 11. What the GA fixed: archetype reshuffle =====================
\begin{frame}{What the GA fixed: which archetypes win more (and less)?}
  \centerline{\includegraphics[width=\linewidth,height=0.58\textheight,keepaspectratio]{fig_archetype_shift.png}}
  \vspace{0.2em}
  \begin{itemize}\footnotesize\itemsep=0.22em
    \item \textbf{GA winner reshuffles \emph{who} wins, not just shortens:}
      \begin{itemize}\scriptsize\itemsep=0.15em
        \item Slow / cash-conserving (CashHoarder, RiskAverse, Passive): middling $\rightarrow$ top-tier.
      \end{itemize}
    \item Bad strategies perform the same across both boards: performance is a function of strategy \emph{and} board type.
    \item \textbf{Takeaway:} we can use our method to determine if a strategy is truly good.
  \end{itemize}
\end{frame}

% ===================== 12. Results: generalization to 2-8 players =====================
\begin{frame}{Results: generalization to 2--8 players}
  \centerline{\includegraphics[width=\linewidth,height=0.62\textheight,keepaspectratio]{fig_cross_eval_scaling.png}}
  \vspace{0.2em}
  \begin{itemize}\footnotesize\itemsep=0.28em
    \item \textbf{Takeaway:} every metric except \emph{fairness} generalises from training (2p / 3p) to every count from $2$ to $8$; fairness is \emph{strongly dependent on the number of players}.
    \item \textbf{Game Length:} $\sim$$60$ rounds at every count vs.\ $104 \!\to\! 158$ for the default.
    \item \textbf{Decisiveness:} optimised boards finish $\geq 78\%$ of games at 8p; the default finishes only $\mathbf{29\%}$.
    \item \textbf{Fairness:} winners are $30$--$53\%$ \emph{fairer} than default at 2p / 3p but $69$--$101\%$ \emph{less} fair at 7--8p.
  \end{itemize}
\end{frame}

% ===================== 13. Conclusion =====================
\begin{frame}{Conclusion}
  \begin{itemize}\footnotesize\itemsep=0.55em
    \item \textbf{Human playtest ($N{=}10$, two players) clears the success bar.}
    \item \textbf{Pool $+$ LLM evaluators agree on the rule-based GA winner.}
    \item Optimised boards play \textbf{$\sim$2--3$\times$ faster}, finish reliably at \textbf{2--8 players}.
    \item We can use our framework to analyse game-board characteristics and player behaviour.
    \item The recipe transfers to any parameterised multi-agent system: market mechanisms, RL environments, recommendation systems.
  \end{itemize}
\end{frame}

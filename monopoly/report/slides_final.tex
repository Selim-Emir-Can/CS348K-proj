% ===================== 2. Motivation =====================
\begin{frame}{Motivation}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.58\textwidth}
      \begin{itemize}\itemsep=0.7em
        \item Beta testing is resource-intensive: balancing a game before release takes many diverse human testers over many sessions.
        \item Game designers face combinatorial design spaces that human playtesting cannot cover.
        \item What is needed: an automated, reproducible way to test each candidate design against a \emph{diverse population} of simulated players, not just one fixed test-player.
      \end{itemize}
    \end{column}
    \begin{column}{0.4\textwidth}
      \centering
      \includegraphics[width=\linewidth]{board_default.png}\\
      \tiny Monopoly: our testbed
    \end{column}
  \end{columns}
\end{frame}

% ===================== 3. The problem, and why it is hard =====================
\begin{frame}{The problem, and why it is hard}
  \begin{itemize}\itemsep=0.7em
    \item How can we accelerate beta testing while ensuring that every playstyle enjoys the game, across a combination of metrics?
    \item \textbf{Prior automated playtesting evaluates with a single agent class:}
      \begin{itemize}\itemsep=0.2em
        \item a single trained LLM player,
        \item \emph{procedural personas}: learned agents matched to human playstyles, deterministic dynamics,
        \item Isaksen et al.\ (2018): one \emph{skill model}, varies skill level (modes of play not modeled).
      \end{itemize}
    \item \textbf{Why they fall short:} one agent class cannot detect its own blind spots; even a diverse pool can rate a design fair while a different class sees it as broken.
  \end{itemize}
\end{frame}

\begin{frame}{Our approach}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.6\textwidth}
      \begin{itemize}\itemsep=0.6em
        \item \textbf{Closed loop:} a genetic algorithm searches a 66-dim board design space to minimise a 5-axis composite objective.
        \item \textbf{Evaluate} each candidate board against a fixed pool of 30 diverse rule-based strategies (10 archetypes + 20 sampled).
        \item \textbf{Cross-class check:} re-evaluate the winners with an \emph{independent} agent class (a small LLM, Qwen2.5-1.5B).
        \item \textbf{Human pilot:} play the winning boards with people to check the predicted pacing and money-flow effects hold.
      \end{itemize}
    \end{column}

% ===================== 5. The objective =====================
\begin{frame}{The objective: one composite, five design qualities}
  \[
    \text{score}(\theta) \;=\; w_F\,\overline{F} \;+\; w_{F_{\max}}\,F_{\max}
    \;+\; w_R\,\tfrac{|\overline{R}-R^*|}{R^*} \;+\; w_D\,\overline{D}
    \;+\; w_T\,\tfrac{|\overline{T}-T^*|}{T^*}
  \]
  \vspace{0.2em}
  \begin{center}
  \small
  \begin{tabular}{lll}
    \toprule
    Term & Measures & Goal \\
    \midrule
    $\overline{F}$ mean fairness     & win-rate spread between players (avg.\ over matchups) & minimise \\
    $F_{\max}$ worst-pair fairness   & the \emph{worst} matchup's spread (robustness)        & minimise \\
    $\overline{R}$ length            & mean rounds per game                                  & target $60$ \\
    $\overline{D}$ draw rate         & fraction of games with no winner                      & minimise \\
    $\overline{T}$ transfer          & inter-player money moved per player-turn              & target $50$ \\
    \bottomrule
  \end{tabular}
  \end{center}
  \vspace{0.4em}
  \begin{itemize}\small
    \item Zeroing terms one at a time gives the single-objective ablations.
  \end{itemize}
\end{frame}

% ===================== Results: generalization to 2-8 players =====================
\begin{frame}{Results: generalization to 2--8 players}
  \centerline{\includegraphics[width=\linewidth,height=0.62\textheight,keepaspectratio]{fig_cross_eval_scaling.png}}
  \vspace{0.2em}
  \begin{itemize}\footnotesize\itemsep=0.28em
    \item \textbf{Takeaway:} every metric except \emph{fairness} generalises from training (2p / 3p) to every count from $2$ to $8$; fairness is \emph{strongly dependent on the number of players}.
    \item \textbf{Game Length:} $\sim$$60$ rounds at every count vs.\ $104 \!\to\! 158$ for the default.
    \item \textbf{Decisiveness:} optimised boards finish $\geq 78\%$ of games at 8p; the default finishes only $\mathbf{29\%}$.
    \item \textbf{Fairness:} winners are $30$--$53\%$ \emph{fairer} than default at 2p / 3p but $69$--$101\%$ \emph{less} fair at 7--8p (regime-specific, by construction).
  \end{itemize}
\end{frame}

% ===================== Results: real human play =====================
\begin{frame}{Does it hold up with a human at the controls?}
  \centerline{\includegraphics[width=\linewidth,height=0.52\textheight,keepaspectratio]{fig_human_playtest.png}}
  \vspace{0.3em}
  \begin{itemize}\small\itemsep=0.3em
    \item \textbf{Pilot:} two players, two 2p boards (default + GA winner), each played three times under \emph{matched} seeds (common random numbers); $N{=}3$, all four metrics in the figure are cited directionally including fairness.
    \item \textbf{Pacing holds:} GA winner finishes in $\sim$29 rounds vs.\ \textbf{122} for the default, which once hit the 200-turn cap with \emph{no winner}; all 3 GA-winner games ended decisively, the default only 2 of 3.
    \item \textbf{Money circulates more, draws drop, balance improves:} $68$ \$/round vs.\ $43$; $0\%$ draw rate vs.\ $33\%$; win-rate spread $0.33$ vs.\ $0.67$. Directional evidence that all four trained effects survive contact with real players.
  \end{itemize}
\end{frame}

% ===================== Takeaway =====================
\begin{frame}{Takeaway}
  \begin{itemize}\footnotesize\itemsep=0.55em
    \item Optimised boards play \textbf{$\sim$2--3$\times$ faster}, finish reliably at \textbf{2--8 players}, and maintain a clean $\sim$$2\times$ composite advantage at every count under cross-count-comparable normalisation, even though the GA only trained at 2p / 3p.
    \item \textbf{Fairness}, by contrast, is \emph{strongly dependent on the number of players}: the winners are fairer than default in their training regime but \emph{less} fair at 6--8 players.
    \item \textbf{Two evaluator classes (rule-based pool + LLM seats) agree on the rule-based GA winner}, and rate it 4/5 metrics over an LLM-driven board: the pool generalises across many play styles where LLM-only training overfits to its narrow decision distribution, and costs nothing per decision.
    \item A small $N{=}2$ human playtest pilot confirms the predicted pacing and money-flow effects survive contact with real players.
    \item The recipe (closed-loop GA $+$ diverse evaluator pool $+$ independent cross-class checks) transfers to any parameterised multi-agent system: market mechanisms, RL environments, recommendation systems.
  \end{itemize}
  \vspace{0.7em}
  \begin{center}
    \small Code, logs, figures: \url{github.com/Selim-Emir-Can/CS348K-proj/tree/master}
  \end{center}
\end{frame}

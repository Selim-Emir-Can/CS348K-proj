% ===================== 2. Motivation =====================
\begin{frame}{Motivation}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.58\textwidth}
      \begin{itemize}\itemsep=0.7em
        \item Beta testing is resource-intensive; balance a game $\rightarrow$ many diverse human testers over many sessions.
        \item Combinatorial design spaces that human playtesting alone cannot cover.
        \item We need an automated way to test each candidate design against a \emph{diverse population} of simulated players.
      \end{itemize}
    \end{column}
    \begin{column}{0.4\textwidth}
      \centering
      \includegraphics[width=\linewidth]{board_default.png}\\
      \tiny Monopoly: our testbed
    \end{column}
  \end{columns}
\end{frame}

% ===================== 3. The problem, and why it is hard =====================
\iffalse
\begin{frame}{Problem \& Prior Work}
  \begin{itemize}\itemsep=0.7em
    \item How can we accelerate beta testing while ensuring that every playstyle enjoys the game, across a combination of metrics?
    \item \textbf{Prior automated playtesting evaluates with a single agent class:}
      \begin{itemize}\itemsep=0.2em
        \item a single trained LLM player,
        \item \emph{procedural personas}: learned agents matched to human playstyles, deterministic dynamics,
        \item Isaksen et al.\ (2018): one \emph{skill model}, varies skill level (modes of play not modeled).
      \end{itemize}

  \end{itemize}
\end{frame}

% ===================== 4. Our approach (HIDDEN: the method slide covers it) =====================
\iffalse
\begin{frame}{Our approach}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.6\textwidth}
      \begin{itemize}\itemsep=0.6em
        \item \textbf{Closed loop:} a genetic algorithm searches a 66-dim board design space to minimise a 5-axis composite objective.
        \item \textbf{Evaluate} each candidate board against a fixed pool of 30 diverse rule-based strategies (10 archetypes + 20 sampled).
        \item \textbf{Cross-class check:} re-evaluate the winners with an \emph{independent} agent class (a small LLM, Qwen2.5-1.5B).
        \item \textbf{Human pilot:} play the winning boards with people to check the predicted pacing and money-flow effects hold.
      \end{itemize}
    \end{column}
    \begin{column}{0.4\textwidth}

% ===================== 5. The objective =====================
\begin{frame}{The objective: one composite, five design qualities}
  \[
    \text{score}(\theta) \;=\; w_F\,\overline{F} \;+\; w_{F_{\max}}\,F_{\max}
    \;+\; w_R\,\tfrac{|\overline{R}-R^*|}{R^*} \;+\; w_D\,\overline{D}
    \;+\; w_T\,\tfrac{|\overline{T}-T^*|}{T^*}
  \]
  \vspace{0.2em}
  \begin{center}
  \small
  \begin{tabular}{lll}
    \toprule
    Term & Measures & Goal \\
    \midrule
    $\overline{F}$ mean fairness     & win-rate spread between players (avg.\ over matchups) & minimise \\
    $F_{\max}$ worst-pair fairness   & the \emph{worst} matchup's spread (robustness)        & minimise \\
    $\overline{R}$ length            & mean rounds per game                                  & target $60$ \\
    $\overline{D}$ draw rate         & fraction of games with no winner                      & minimise \\
    $\overline{T}$ transfer          & inter-player money moved per player-turn              & target $50$ \\
    \bottomrule
  \end{tabular}
  \end{center}
  \vspace{0.4em}
  \begin{itemize}\small
    \item Zeroing terms one at a time gives the single-objective ablations.
  \end{itemize}
\end{frame}

% ===================== Game-design insights =====================
\begin{frame}{Game-design insights}
  \begin{itemize}\footnotesize\itemsep=0.35em
    \item \textbf{Expansion $\rightarrow$ solvency: losing condition flips.}
      \begin{itemize}\scriptsize\itemsep=0.15em
        \item Default (long games, low rents): reward owning more.
        \item GA winner ($\sim$$60$ rounds, $\sim$$1.3\times$ rent, $\sim$$1.6\times$ cost): reward staying solvent.
        \item One bad landing can drain a \$$300$ reserve.
        \item AggressiveBuilder vs.\ CashHoarder: $0.80\!\to\!0.45$ (WR).
      \end{itemize}
    \item \textbf{Smaller colour groups make all-in builds work.}
      \begin{itemize}\scriptsize\itemsep=0.15em
        \item Default: cheap 2-cell groups (Brown) can't win; need an expensive triplet.
        \item GA winner: $1$--$2$ cell groups $+$ $1.6\times$ cost / $1.3\times$ rent $\rightarrow$ max houses $+$ hotel pays off.
      \end{itemize}
    \item \textbf{Blocking opponents stops working.}
      \begin{itemize}\scriptsize\itemsep=0.15em
        \item Default: buy one cell of opponent's set to deny completion.
        \item GA winner: $1$--$2$ cell sets complete on first landing.
        \item Higher costs ($\sim$$1.6\times$) prevent hoarding properties before crossing GO.
        \item Something rent tuning alone could not do.
      \end{itemize}
    \item \textbf{Game length} ($\sim$$60$ vs.\ $\sim$$100$ rounds): the $12$-cell lap leaves fewer turns between rent events.
    \item \textbf{Decisiveness} (draw rate $7\%\!\to\!2\%$): $\sim$$1.6\times$ base cost and $\sim$$1.3\times$ base rent force bankruptcy before the turn cap.
    \item \textbf{Money circulation} (\$$50\!\to\!$\$$73$/round): higher base rent on cells landed more often.
    \item \textbf{Fairness at 2p/3p} (spread $0.45\!\to\!0.22$; $-32\%$ at 3p): Yellow dropped $+$ most groups shrunk to $1$--$2$ cells $+$ per-property rent multipliers spread $\sim$$0.6\times$--$1.9\times$ damp over-powered cells, the same recipe the fairness-only ablation finds.
  \end{itemize}
\end{frame}

% ===================== Results: archetype win-rate reshuffle =====================
\begin{frame}{What the GA fixed: which archetypes win more (and less)?}
  \centerline{\includegraphics[width=\linewidth,height=0.60\textheight,keepaspectratio]{fig_archetype_shift.png}}
  \vspace{0.2em}
  \begin{itemize}\footnotesize\itemsep=0.22em
    % \item WR vs.\ field $=$ mean win-rate against the other $29$ pool strategies ($20$ games per pair, $580$ games/archetype/board).
    \item \textbf{GA winner reshuffles \emph{who} wins, not just shortens:}
      \begin{itemize}\scriptsize\itemsep=0.15em
        \item Slow / cash-conserving (CashHoarder, RiskAverse, Passive): middling $\rightarrow$ top-tier.
        \item HighCostOnly: collapses, keep-mask prunes the expensive groups it relies on.
      \end{itemize}
    \item Bad strategies perform the same across both boards: performance is a function of strategy \emph{and} board type.
    \item Top \emph{and} bottom are board-robust; reshuffle concentrated in the \emph{middle} (assuming similar dynamics).
  \end{itemize}
\end{frame}

% ===================== Results: generalization to 2-8 players =====================
\begin{frame}{Results: generalization to 2--8 players}
  \centerline{\includegraphics[width=\linewidth,height=0.62\textheight,keepaspectratio]{fig_cross_eval_scaling.png}}
  \vspace{0.2em}
  \begin{itemize}\footnotesize\itemsep=0.28em
    \item \textbf{Takeaway:} every metric except \emph{fairness} generalises from training (2p / 3p) to every count from $2$ to $8$; fairness is \emph{strongly dependent on the number of players}.
    \item \textbf{Game Length:} $\sim$$60$ rounds at every count vs.\ $104 \!\to\! 158$ for the default.
    \item \textbf{Decisiveness:} optimised boards finish $\geq 78\%$ of games at 8p; the default finishes only $\mathbf{29\%}$.
    \item \textbf{Fairness:} winners are $30$--$53\%$ \emph{fairer} than default at 2p / 3p but $69$--$101\%$ \emph{less} fair at 7--8p (regime-specific, by construction).
  \end{itemize}
\end{frame}

% ===================== Conclusion =====================
\begin{frame}{Conclusion}
  \begin{itemize}\footnotesize\itemsep=0.55em
    \item \textbf{Human playtest ($N{=}10$, two players) clears the success bar:}
      % \begin{itemize}\scriptsize\itemsep=0.15em
      %   \item Constraints specified in the agent setting are satisfied in real human play ($\sim$$50$ vs.\ $131$ rounds, \$$66$ vs.\ \$$39$/round, $0\%$ vs.\ $40\%$ draws); direction $+$ magnitude hold across pool, LLM, and human.
      %   \item Agent feedback is conservative, not optimistic: humans show a \emph{larger} default failure ($40\%$ draws) than either simulator ($0$--$7\%$).
      % \end{itemize}
    \item Optimised boards play \textbf{$\sim$2--3$\times$ faster}, finish reliably at \textbf{2--8 players}, and maintain a clean $\sim$$2\times$ composite advantage, even though the GA only trained at 2p / 3p.
    \item \textbf{Pool $+$ LLM evaluators agree on the rule-based GA winner.}
      % \begin{itemize}\scriptsize\itemsep=0.15em
      %   \item Pool-driven board beats LLM-driven board, $4/5$ metrics, under \emph{both} evaluators.
      %   \item Pool generalises to unseen evaluator classes; LLM-only overfits its narrow decisions.
      %   \item Pool is essentially free, no per-decision LLM call.
      % \end{itemize}
    \item Some metrics depend on number of players (e.g.\ fairness); performance is a function of player count, strategy space, \emph{and} board.
    \item The recipe (closed-loop GA $+$ diverse evaluator pool $+$ independent cross-class checks) transfers to any parameterised multi-agent system: market mechanisms, RL environments, recommendation systems.
  \end{itemize}
\end{frame}
